import random import tkinter as tk from tkinter import messagebox class MathGame: def __init__(self, root): self.root = root self.root.title("🎯 Clearfield South Elementary Math Challenge!") self.root.geometry("600x500") self.root.configure(bg="#2C3E50") # Deep playful blue background # Game variables self.score = 0 self.lives = 3 self.num1 = 0 self.num2 = 0 self.correct_answer = 0 # Colors for UI self.bg_color = "#2C3E50" self.card_color = "#ECF0F1" self.accent_color = "#E67E22" self.success_color = "#2ECC71" self.fail_color = "#E74C3C" self.setup_ui() self.next_question() def setup_ui(self): # Header Canvas/Frame for Score & Lives self.stats_frame = tk.Frame(self.root, bg=self.bg_color) self.stats_frame.pack(pady=20, fill="x") self.score_label = tk.Label( self.stats_frame, text=f"⭐ Score: {self.score}", font=("Arial", 18, "bold"), fg="#F1C40F", bg=self.bg_color, ) self.score_label.pack(side="left", padx=40) self.lives_label = tk.Label( self.stats_frame, text=f"❤️ Lives: {'❤️' * self.lives}", font=("Arial", 18, "bold"), fg=self.fail_color, bg=self.bg_color, ) self.lives_label.pack(side="right", padx=40) # Main Question Card self.card = tk.Frame( self.root, bg=self.card_color, bd=5, relief="groove" ) self.card.pack(pady=10, padx=40, fill="both", expand=True) self.title_label = tk.Label( self.card, text="The Times 12 Challenge!", font=("Arial", 22, "bold"), fg="#34495E", bg=self.card_color, ) self.title_label.pack(pady=15) # Large Equation display self.equation_label = tk.Label( self.card, text="", font=("Arial", 48, "bold"), fg=self.accent_color, bg=self.card_color, ) self.equation_label.pack(pady=10) # Answer Entry Box self.answer_var = tk.StringVar() self.entry = tk.Entry( self.card, textvariable=self.answer_var, font=("Arial", 24, "bold"), justify="center", width=8, bd=3, ) self.entry.pack(pady=10) self.entry.bind("", lambda event: self.check_answer()) self.entry.focus() # Submit Button self.submit_btn = tk.Button( self.card, text="SUBMIT!", command=self.check_answer, font=("Arial", 16, "bold"), bg=self.success_color, fg="white", activebackground="#27AE60", activeforeground="white", cursor="hand2", padx=20, pady=5, ) self.submit_btn.pack(pady=15) # Feedback Label self.feedback_label = tk.Label( self.card, text="Type your answer and press enter!", font=("Arial", 14, "italic"), fg="#7F8C8D", bg=self.card_color, ) self.feedback_label.pack(pady=10) def next_question(self): # Pick two numbers between 1 and 12 self.num1 = random.randint(1, 12) self.num2 = random.randint(1, 12) self.correct_answer = self.num1 * self.num2 # Update Equation display self.equation_label.config(text=f"{self.num1} × {self.num2} = ?") self.answer_var.set("") self.entry.focus() def check_answer(self): user_input = self.answer_var.get().strip() # Input validation for younger students if not user_input.isdigit(): self.feedback_label.config( text="Oops! Please type a number!", fg=self.fail_color ) return user_answer = int(user_input) if user_answer == self.correct_answer: self.score += 10 self.score_label.config(text=f"⭐ Score: {self.score}") # Fun randomized success messages success_messages = [ "🎉 Awesome Job!", "🚀 You are a Math Superstar!", "💯 Correct! Keep it up!", "🌟 High Five! That's right!", ] self.feedback_label.config( text=random.choice(success_messages), fg=self.success_color ) self.root.after(1200, self.next_question) else: self.lives -= 1 self.lives_label.config(text=f"❤️ Lives: {'❤️' * self.lives}") if self.lives <= 0: self.game_over() else: self.feedback_label.config( text=f"❌ Not quite! Try the next one!", fg=self.fail_color ) self.root.after(1500, self.next_question) def game_over(self): messagebox.showinfo( "Game Over! 🏁", f"Great effort! You scored {self.score} points!\nLet's try again to beat your high score!", ) # Reset game parameters self.score = 0 self.lives = 3 self.score_label.config(text=f"⭐ Score: {self.score}") self.lives_label.config(text=f"❤️ Lives: {'❤️' * self.lives}") self.feedback_label.config( text="Type your answer and press enter!", fg="#7F8C8D" ) self.next_question() if __name__ == "__main__": root = tk.Tk() game = MathGame(root) root.mainloop()
